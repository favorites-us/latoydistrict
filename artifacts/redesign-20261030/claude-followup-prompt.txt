import "server-only";
import { database, PHOTO_BUCKET } from "./supabase";
import type { StorePhoto } from "@/lib/photos";

export async function photoIsReported(photo: StorePhoto) {
  const db = database();
  if (!db) throw new Error("storage_unavailable");
  // Permanent anonymous suppression records are separate from request inbox
  // rows. Marking a request handled or editing its notes never unblocks a photo.
  const { data, error } = await db.from("leads").select("id").eq("buyer_type", "photo_block").eq("store_name", photo.id).limit(1);
  if (error) throw new Error("report_check_failed");
  return !!data?.length;
}

export async function removePhoto(photo: StorePhoto, report: { name: string; contact: string; reason: string; locale: string }) {
  const db = database();
  if (!db) throw new Error("storage_unavailable");
  // Persist the block before deleting bytes. Every image request checks this
  // record, including requests to older deployments. Failure must stay closed.
  const blockId = crypto.randomUUID();
  const blockState = { photo_id: photo.id, bytes_removed: false };
  // A bulk insert is transactional: either both the anonymous block and the
  // private request are persisted, or neither is. No schema migration needed.
  const { error } = await db.from("leads").insert([
    { id: blockId, name: "Photo suppression", contact: "", items: JSON.stringify(blockState), locale: "en", buyer_type: "photo_block", store_slug: photo.store_slug, store_name: photo.id, notes: "Permanent suppression record. Keep when archiving the request inbox.", handled: false },
    { name: report.name, contact: report.contact, items: report.reason, locale: report.locale, buyer_type: "photo_removal", store_slug: photo.store_slug, store_name: photo.source_label, notes: `photo_id:${photo.id}`, handled: false },
  ]);
  if (error) throw new Error("report_save_failed");
  const { error: deletionError } = await db.storage.from(PHOTO_BUCKET).remove([photo.storage_path]);
  // Best-effort operational audit. An audit-write failure cannot remove a block.
  await db.from("leads").update({ items: JSON.stringify({ ...blockState, bytes_removed: !deletionError }) }).eq("id", blockId);
  return { hidden: true, bytes_removed: !deletionError };
}

import { NextResponse } from "next/server";
import { photoById } from "@/lib/photos";
import { database, PHOTO_BUCKET } from "@/lib/server/supabase";
import { photoIsReported, removePhoto } from "@/lib/server/photo-service";
import { limitedJson, textField } from "@/lib/request-validation";

export const runtime = "nodejs";
export const maxDuration = 30;
export async function POST(request: Request) {
  const configured = process.env.PHOTO_ADMIN_TOKEN;
  const supplied = request.headers.get("authorization")?.replace(/^Bearer /, "") ?? "";
  if (!configured || configured.length < 32 || !supplied || !timingSafeEqual(createHash("sha256").update(configured).digest(), createHash("sha256").update(supplied).digest())) return NextResponse.json({ ok: false }, { status: 401 });
  const body = await limitedJson(request, 3_000_000);
  const photo = body && photoById(textField(body.photo_id, 120));
  if (!body || !photo) return NextResponse.json({ ok: false }, { status: 400 });
  const db = database();
  if (!db) return NextResponse.json({ ok: false, reason: "storage_unavailable" }, { status: 503 });
  try {
    if (body.action === "check") {
      // Bounded integration check: no real person, claim, or photo is affected.
      // Validate the actual table's columns/types/constraints before publishing.
      const ids = [crypto.randomUUID(), crypto.randomUUID()];
      const rows = ["photo_block", "photo_removal"].map((buyer_type, index) => ({ id: ids[index], name: "SYSTEM: photo schema check", contact: "", buyer_type, items: "Temporary schema check. No removal request or ownership claim.", locale: "en", store_slug: null, store_name: `__photo_schema_check_${ids[0]}`, handled: true, notes: "Temporary system diagnostic; safe to delete." }));
      const { error: insertError } = await db.from("leads").insert(rows);
      if (insertError) return NextResponse.json({ ok: false, reason: "report_schema_unavailable" }, { status: 503 });
      const { error: cleanupError } = await db.from("leads").delete().in("id", ids);
      if (cleanupError) return NextResponse.json({ ok: false, reason: "schema_check_cleanup_failed", diagnostic_ids: ids }, { status: 503 });
      const { data, error: readError } = await db.from("leads").select("id").in("id", ids);
      if (readError || data?.length) return NextResponse.json({ ok: false, reason: "schema_check_cleanup_unconfirmed", diagnostic_ids: ids }, { status: 503 });
      return NextResponse.json({ ok: true, report_schema: "validated", diagnostic_rows_remaining: 0 });
    }
    if (body.action === "remove") {
      const result = await removePhoto(photo, { name: "Directory operator", contact: "", reason: textField(body.reason, 1500) || "Removed by operator", locale: "en" });
      return NextResponse.json({ ok: true, ...result });
    }
    if (body.action !== "publish" || photo.status !== "published" || typeof body.image_base64 !== "string") return NextResponse.json({ ok: false }, { status: 400 });
    if (await photoIsReported(photo)) return NextResponse.json({ ok: false, reason: "removal_request_pending" }, { status: 409 });
    const bytes = Buffer.from(body.image_base64, "base64");
    if (!bytes.length || bytes.length > 2_000_000 || createHash("sha256").update(bytes).digest("hex") !== photo.sha256) return NextResponse.json({ ok: false, reason: "unreviewed_bytes" }, { status: 400 });
    const { data: bucket, error: bucketError } = await db.storage.getBucket(PHOTO_BUCKET);
    if (bucket?.public) return NextResponse.json({ ok: false, reason: "bucket_must_be_private" }, { status: 409 });
    if (!bucket) {
      const { error } = await db.storage.createBucket(PHOTO_BUCKET, { public: false, fileSizeLimit: 2_000_000, allowedMimeTypes: ["image/jpeg", "image/png", "image/webp"] });
      if (error) return NextResponse.json({ ok: false, reason: bucketError ? "bucket_unavailable" : "bucket_create_failed" }, { status: 503 });
    }
    const { error } = await db.storage.from(PHOTO_BUCKET).upload(photo.storage_path, bytes, { contentType: photo.mime_type, cacheControl: "0", upsert: false });
    if (error) return NextResponse.json({ ok: false, reason: "upload_failed_or_exists" }, { status: 409 });
    // A report racing with an upload remains blocked and is never auto-restored.
    if (await photoIsReported(photo)) { await db.storage.from(PHOTO_BUCKET).remove([photo.storage_path]); return NextResponse.json({ ok: false, reason: "removal_request_pending" }, { status: 409 }); }
    return NextResponse.json({ ok: true, photo_id: photo.id });
  } catch { return NextResponse.json({ ok: false, reason: "storage_error" }, { status: 503 }); }
}

  if (!photo || photo.status !== "published") return new Response(null, { status: 404, headers });

    if (await photoIsReported(photo)) return new Response(null, { status: 410, headers });

  } catch { return new Response(null, { status: 503, headers }); }
