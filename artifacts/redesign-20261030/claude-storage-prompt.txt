import "server-only";
import { del, get, put } from "@vercel/blob";

// All files belong to this site's private store. Never expose storage URLs or
// credentials. Suppression reads must bypass the CDN, including cached misses.
export const privateFiles = {
  async read(path: string): Promise<Buffer | null> {
    const result = await get(path, { access: "private", useCache: false, abortSignal: AbortSignal.timeout(12000) });
    if (!result) return null;
    if (result.statusCode !== 200 || !result.stream) throw new Error("private_read_failed");
    return Buffer.from(await new Response(result.stream).arrayBuffer());
  },
  async write(path: string, bytes: string | Buffer, contentType = "application/json", overwrite = false) {
    const result = await put(path, bytes, { access: "private", addRandomSuffix: false, allowOverwrite: overwrite, contentType, cacheControlMaxAge: 60, abortSignal: AbortSignal.timeout(12000) });
    if (!new URL(result.url).hostname.endsWith(".private.blob.vercel-storage.com")) throw new Error("private_store_required");
  },
  async remove(path: string) { await del(path, { abortSignal: AbortSignal.timeout(12000) }); },
};

export async function checkPrivateStorage() {
  const path = `diagnostics/${crypto.randomUUID()}.json`;
  try {
    await privateFiles.write(path, JSON.stringify({ diagnostic: "storage check", revision: 1 }));
    if (JSON.parse((await privateFiles.read(path))?.toString() ?? "null")?.revision !== 1) throw new Error("write_read_failed");
    await privateFiles.write(path, JSON.stringify({ diagnostic: "storage check", revision: 2 }), "application/json", true);
    if (JSON.parse((await privateFiles.read(path))?.toString() ?? "null")?.revision !== 2) throw new Error("consistent_read_failed");
  } finally { await privateFiles.remove(path); }
  if (await privateFiles.read(path)) throw new Error("cleanup_unconfirmed");
}

import "server-only";
import { privateFiles } from "./private-files";
import type { StorePhoto } from "@/lib/photos";

type RemovalRequest = { name: string; contact: string; reason: string; locale: string };
export function photoBlockPath(photo: StorePhoto) { return `photo-blocks/${photo.id}.json`; }
export async function photoIsReported(photo: StorePhoto) {
  // Existence is the permanent block. Request handling and inbox deletion do
  // not affect it. Read errors propagate, so image serving fails closed.
  return (await privateFiles.read(photoBlockPath(photo))) !== null;
}

export async function removePhoto(photo: StorePhoto, report: RemovalRequest) {
  const id = crypto.randomUUID();
  const created_at = new Date().toISOString();
  // Persist anonymous suppression first. Its only meaning is "never serve".
  await privateFiles.write(photoBlockPath(photo), JSON.stringify({ photo_id: photo.id, blocked_at: created_at }), "application/json", true);
  let request_saved = false;
  try {
    await privateFiles.write(`photo-requests/${id}.json`, JSON.stringify({ id, created_at, photo_id: photo.id, ...report }));
    request_saved = true;
  } catch { /* Removing the photo takes priority over storing contact details. */ }
  let bytes_removed = false;
  try { await privateFiles.remove(photo.storage_path); bytes_removed = true; } catch { /* Persisted block still prevents serving. */ }
  try { await privateFiles.write(`photo-audits/${id}.json`, JSON.stringify({ photo_id: photo.id, created_at, bytes_removed, request_saved })); } catch { /* Block remains permanent even if audit storage fails. */ }
  return { hidden: true, bytes_removed, request_saved };
}

import { createHash, timingSafeEqual } from "node:crypto";
import { NextResponse } from "next/server";
import { photoById } from "@/lib/photos";
import { privateFiles, checkPrivateStorage } from "@/lib/server/private-files";
import { photoIsReported, removePhoto } from "@/lib/server/photo-service";
import { limitedJson, textField } from "@/lib/request-validation";

export const runtime = "nodejs";
export const maxDuration = 60;
export async function POST(request: Request) {
  const configured = process.env.PHOTO_ADMIN_TOKEN;
  const supplied = request.headers.get("authorization")?.replace(/^Bearer /, "") ?? "";
  if (!configured || configured.length < 32 || !supplied || !timingSafeEqual(createHash("sha256").update(configured).digest(), createHash("sha256").update(supplied).digest())) return NextResponse.json({ ok: false }, { status: 401 });
  const body = await limitedJson(request, 3_000_000);
  const photo = body && photoById(textField(body.photo_id, 120));
  if (!body || !photo) return NextResponse.json({ ok: false }, { status: 400 });
  try {
    if (body.action === "check") {
      await checkPrivateStorage();
      return NextResponse.json({ ok: true, private_storage: "validated", diagnostic_files_remaining: 0 });
    }
    if (body.action === "remove") {
      const result = await removePhoto(photo, { name: "Directory operator", contact: "", reason: textField(body.reason, 1500) || "Removed by operator", locale: "en" });
      return NextResponse.json({ ok: true, ...result });
    }
    if (body.action !== "publish" || photo.status !== "published" || typeof body.image_base64 !== "string") return NextResponse.json({ ok: false }, { status: 400 });
    if (await photoIsReported(photo)) return NextResponse.json({ ok: false, reason: "removal_request_pending" }, { status: 409 });
    const bytes = Buffer.from(body.image_base64, "base64");
    if (!bytes.length || bytes.length > 2_000_000 || createHash("sha256").update(bytes).digest("hex") !== photo.sha256) return NextResponse.json({ ok: false, reason: "unreviewed_bytes" }, { status: 400 });
    await privateFiles.write(photo.storage_path, bytes, photo.mime_type);
    if (await photoIsReported(photo)) { await privateFiles.remove(photo.storage_path); return NextResponse.json({ ok: false, reason: "removal_request_pending" }, { status: 409 }); }
    return NextResponse.json({ ok: true, photo_id: photo.id });
  } catch { return NextResponse.json({ ok: false, reason: "private_storage_error" }, { status: 503 }); }
}

import { photoById } from "@/lib/photos";
import { privateFiles } from "@/lib/server/private-files";
import { photoIsReported } from "@/lib/server/photo-service";

export const dynamic = "force-dynamic";
export const runtime = "nodejs";
const headers = { "Cache-Control": "private, no-store, max-age=0", "CDN-Cache-Control": "no-store", "Vercel-CDN-Cache-Control": "no-store", "X-Robots-Tag": "noindex, noimageindex", "X-Content-Type-Options": "nosniff" };

export async function GET(_request: Request, { params }: { params: Promise<{ id: string }> }) {
  const { id } = await params;
  const photo = photoById(id);
  if (!photo || photo.status !== "published") return new Response(null, { status: 404, headers });
  try {
    if (await photoIsReported(photo)) return new Response(null, { status: 410, headers });
    const bytes = await privateFiles.read(photo.storage_path);
    if (!bytes) return new Response(null, { status: 404, headers });
    if (await photoIsReported(photo)) return new Response(null, { status: 410, headers });
    return new Response(new Uint8Array(bytes), { headers: { ...headers, "Content-Type": photo.mime_type } });
  } catch { return new Response(null, { status: 503, headers }); }
}

import { NextResponse } from "next/server";
import { photoById } from "@/lib/photos";
import { limitedJson, sameOrigin, textField, validPhotoReport } from "@/lib/request-validation";
import { photoIsReported, removePhoto } from "@/lib/server/photo-service";

export const runtime = "nodejs";
export async function POST(request: Request) {
  if (!sameOrigin(request)) return NextResponse.json({ ok: false }, { status: 403 });
  const body = await limitedJson(request);
  if (!body) return NextResponse.json({ ok: false }, { status: 400 });
  const photo = photoById(textField(body.photo_id, 120));
  const report = validPhotoReport(body);
  if (!photo || !report) return NextResponse.json({ ok: false }, { status: 400 });
  try {
    // Idempotent: repeated reports cannot grow the inbox after a photo is hidden.
    if (await photoIsReported(photo)) return NextResponse.json({ ok: true, hidden: true });
    const result = await removePhoto(photo, report);
    return NextResponse.json({ ok: true, hidden: result.hidden });
  } catch { return NextResponse.json({ ok: false }, { status: 503 }); }
}

import { NextResponse } from "next/server";
import { storeBySlug } from "@/lib/stores";
import { limitedJson, sameOrigin, textField } from "@/lib/request-validation";
import { privateFiles } from "@/lib/server/private-files";

export async function POST(request: Request) {
  if (!sameOrigin(request)) return NextResponse.json({ ok: false, reason: "origin" }, { status: 403 });
  const body = await limitedJson(request);
  if (!body || body.website) return NextResponse.json({ ok: false, reason: "bad_request" }, { status: 400 });
  const slug = textField(body.store_slug, 120);
  const store = slug ? storeBySlug.get(slug) : undefined;
  if (slug && !store) return NextResponse.json({ ok: false, reason: "unknown_store" }, { status: 400 });
  const buyer = textField(body.buyer_type, 40);
  const lead = {
    name: textField(body.name, 120), contact: textField(body.contact, 200),
    buyer_type: ["party_planner", "store_owner", "reseller", "other"].includes(buyer) ? buyer : "other",
    items: textField(body.items, 2000), locale: body.locale === "es" ? "es" : "en",
    store_slug: store?.slug ?? null, store_name: store?.name ?? null,
  };
  if (!lead.name || !lead.contact || !lead.items) return NextResponse.json({ ok: false, reason: "missing_fields" }, { status: 400 });
  try {
    const id = crypto.randomUUID();
    const created_at = new Date().toISOString();
    await privateFiles.write(`inquiries/${created_at.slice(0, 7)}/${id}.json`, JSON.stringify({ id, created_at, ...lead }));
    return NextResponse.json({ ok: true });
  } catch { return NextResponse.json({ ok: false, reason: "storage_error" }, { status: 502 }); }
}
