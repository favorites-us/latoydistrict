"use client";

import { createContext, useContext, useEffect, useRef, useState } from "react";
import { stores } from "@/lib/stores";
import { readShortlist, SHORTLIST_KEY, SHORTLIST_LIMIT } from "@/lib/buyer-tools";
import { track } from "@/lib/analytics";

const known = new Set(stores.map((store) => store.slug));
const Context = createContext<{ saved: string[]; ready: boolean; persistent: boolean; toggle: (slug: string) => boolean }>({ saved: [], ready: false, persistent: true, toggle: () => false });

export function BuyerProvider({ children }: { children: React.ReactNode }) {
  const [saved, setSaved] = useState<string[]>([]);
  const current = useRef<string[]>([]);
  const [ready, setReady] = useState(false);
  const [persistent, setPersistent] = useState(true);
  useEffect(() => {
    const refresh = () => {
      try { current.current = readShortlist(localStorage.getItem(SHORTLIST_KEY), known); setSaved(current.current); }
      catch { setPersistent(false); }
    };
    refresh(); setReady(true);
    const sync = (e: StorageEvent) => { if (e.key === SHORTLIST_KEY || e.key === null) refresh(); };
    window.addEventListener("storage", sync);
    return () => window.removeEventListener("storage", sync);
  }, []);
  function toggle(slug: string) {
    if (!ready || !known.has(slug)) return false;
    const exists = current.current.includes(slug);
    if (!exists && current.current.length >= SHORTLIST_LIMIT) return false;
    const next = exists ? current.current.filter((item) => item !== slug) : [...current.current, slug];
    current.current = next; setSaved(next);
    try { localStorage.setItem(SHORTLIST_KEY, JSON.stringify({ version: 1, slugs: next })); }
    catch { setPersistent(false); }
    track(exists ? "shortlist_remove" : "shortlist_add", { store_slug: slug, count: next.length });
    return true;
  }
  return <Context.Provider value={{ saved, ready, persistent, toggle }}>{children}</Context.Provider>;
}

export function useBuyer() { return useContext(Context); }

export function readShortlist(raw: string | null, knownSlugs: Set<string>): string[] {
  try {
    const parsed = JSON.parse(raw ?? "null");
    if (parsed?.version !== 1 || !Array.isArray(parsed.slugs)) return [];
    return [...new Set<string>(parsed.slugs.filter((slug: unknown) => typeof slug === "string" && knownSlugs.has(slug)))].slice(0, SHORTLIST_LIMIT);
  } catch { return []; }

export function hasBuyingInfo(store: Store): boolean {
  return !!store.online_facts?.some((fact) => fact.source.type === "business");
}

function normalize(value: string): string {
  return value.normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase();
}

export interface DirectoryFilters { query: string; category: string; block: string; info: boolean; contact: boolean }
export const emptyFilters: DirectoryFilters = { query: "", category: "", block: "", info: false, contact: false };

export function filterStores(stores: Store[], filters: DirectoryFilters): Store[] {
  const words = normalize(filters.query.trim()).split(/\s+/).filter(Boolean);
  return stores.filter((store) => {
    if (store.status === "closed") return false;
    if (filters.category && !store.category.some((cat) => cat === filters.category)) return false;
    if (filters.block && store.block !== filters.block) return false;
    if (filters.info && !hasBuyingInfo(store)) return false;
    if (filters.contact && !store.phone && !store.contact_email) return false;
    const haystack = normalize([store.name, store.address.line1, store.address.unit, ...store.category.flatMap((cat) => [catLabel("en", cat), catLabel("es", cat)]), ...(store.online_facts ?? []).filter((fact) => fact.kind === "products").flatMap((fact) => [fact.text.en, fact.text.es])].join(" "));
    return words.every((word) => haystack.includes(word));
  });
}

type BuyerEvent = "shortlist_add" | "shortlist_remove" | "compare_open" | "visit_print" | "quote_copy" | "directory_filter" | "photo_removal_submit" | "store_view" | "store_contact_click" | "lead_submit" | "store_inquiry";

// Buffer early events in the same queue consumed by gtag. Never include free
// text, names, email addresses, phone numbers or quote contents in this queue.
export function track(event: BuyerEvent, params: Record<string, string | number | boolean> = {}) {
  if (typeof window === "undefined" || !process.env.NEXT_PUBLIC_GA_ID) return;
  const w = window as Window & { dataLayer?: unknown[]; gtag?: (...args: unknown[]) => void };
  w.dataLayer ??= [];
  if (w.gtag) w.gtag("event", event, params);
  else {
    function queued(...args: unknown[]) { void args; w.dataLayer!.push(arguments); }
    queued("event", event, params);
  }
}

--- web/lib/request-validation.ts ---
export function sameOrigin(request: Request): boolean {
  const origin = request.headers.get("origin");
  return !!origin && origin === new URL(request.url).origin;
}

export async function limitedJson(request: Request, limit = 12000): Promise<Record<string, unknown> | null> {
  if (!request.headers.get("content-type")?.startsWith("application/json")) return null;
  if (Number(request.headers.get("content-length")) > limit) return null;
  const reader = request.body?.getReader();
  if (!reader) return null;
  const chunks: Uint8Array[] = []; let size = 0;
  try {
    while (true) {
      const { done, value } = await reader.read(); if (done) break;
      size += value.byteLength;
      if (size > limit) { await reader.cancel(); return null; }
      chunks.push(value);
    }
    const bytes = new Uint8Array(size); let offset = 0;
    for (const chunk of chunks) { bytes.set(chunk, offset); offset += chunk.length; }
    const value = JSON.parse(new TextDecoder().decode(bytes));
    return value && typeof value === "object" && !Array.isArray(value) ? value : null;
  } catch { return null; }
}

export function textField(value: unknown, max: number): string {
  return typeof value === "string" ? value.trim().slice(0, max) : "";
}

export function validPhotoReport(body: Record<string, unknown>) {
  const name = textField(body.name, 120);
  const contact = textField(body.contact, 200);
  const reason = textField(body.reason, 1500);
  if (body.website || body.rights_holder !== "yes" || !name || reason.length < 10 || (contact && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(contact))) return null;
  return { name, contact, reason, locale: body.locale === "es" ? "es" : "en" };
}

import { createHash, timingSafeEqual } from "node:crypto";
import { NextResponse } from "next/server";
import { photoById } from "@/lib/photos";
import { database, PHOTO_BUCKET } from "@/lib/server/supabase";
import { photoIsReported, removePhoto } from "@/lib/server/photo-service";
import { limitedJson, textField } from "@/lib/request-validation";

export const runtime = "nodejs";
export const maxDuration = 30;
export async function POST(request: Request) {
  const configured = process.env.PHOTO_ADMIN_TOKEN;
  const supplied = request.headers.get("authorization")?.replace(/^Bearer /, "") ?? "";
  if (!configured || configured.length < 32 || !supplied || !timingSafeEqual(createHash("sha256").update(configured).digest(), createHash("sha256").update(supplied).digest())) return NextResponse.json({ ok: false }, { status: 401 });
  const body = await limitedJson(request, 3_000_000);
  const photo = body && photoById(textField(body.photo_id, 120));
  if (!body || !photo) return NextResponse.json({ ok: false }, { status: 400 });
  const db = database();
  if (!db) return NextResponse.json({ ok: false, reason: "storage_unavailable" }, { status: 503 });
  try {
    if (body.action === "remove") {
      const result = await removePhoto(photo, { name: "Directory operator", contact: "", reason: textField(body.reason, 1500) || "Removed by operator", locale: "en" });
      return NextResponse.json({ ok: true, ...result });
    }
    if (body.action !== "publish" || photo.status !== "published" || typeof body.image_base64 !== "string") return NextResponse.json({ ok: false }, { status: 400 });
    if (await photoIsReported(photo)) return NextResponse.json({ ok: false, reason: "removal_request_pending" }, { status: 409 });
    const bytes = Buffer.from(body.image_base64, "base64");
    if (!bytes.length || bytes.length > 2_000_000 || createHash("sha256").update(bytes).digest("hex") !== photo.sha256) return NextResponse.json({ ok: false, reason: "unreviewed_bytes" }, { status: 400 });
    const { data: bucket, error: bucketError } = await db.storage.getBucket(PHOTO_BUCKET);
    if (bucket?.public) return NextResponse.json({ ok: false, reason: "bucket_must_be_private" }, { status: 409 });
    if (!bucket) {
      const { error } = await db.storage.createBucket(PHOTO_BUCKET, { public: false, fileSizeLimit: 2_000_000, allowedMimeTypes: ["image/jpeg", "image/png", "image/webp"] });
      if (error) return NextResponse.json({ ok: false, reason: bucketError ? "bucket_unavailable" : "bucket_create_failed" }, { status: 503 });
    }
    const { error } = await db.storage.from(PHOTO_BUCKET).upload(photo.storage_path, bytes, { contentType: photo.mime_type, cacheControl: "0", upsert: false });
    if (error) return NextResponse.json({ ok: false, reason: "upload_failed_or_exists" }, { status: 409 });
    // A report racing with an upload remains blocked and is never auto-restored.
    if (await photoIsReported(photo)) { await db.storage.from(PHOTO_BUCKET).remove([photo.storage_path]); return NextResponse.json({ ok: false, reason: "removal_request_pending" }, { status: 409 }); }
    return NextResponse.json({ ok: true, photo_id: photo.id });
  } catch { return NextResponse.json({ ok: false, reason: "storage_error" }, { status: 503 }); }
}

          {nearby.length > 0 ? (
            <>

                {t(locale, "nearby_h2")} {BLOCK_LABELS[store.block]}
              </h2>
              <div className="store-grid">
                {nearby.map((s) => (
                  <StoreCard key={s.slug} store={s} locale={locale} />
                ))}
              </div>
            </>
          ) : null}
        </div>
      </main>
      <Footer locale={locale} />
    </>
  );
}
